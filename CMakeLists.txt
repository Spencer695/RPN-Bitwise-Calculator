cmake_minimum_required(VERSION 3.16)
project(RPNCalculator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(rpn rpn-calculator-example.cpp)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME rpn_regression COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/rpn-cases.csv")
  add_test(NAME reject_short_row COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/short-row.csv")
  set_tests_properties(reject_short_row PROPERTIES WILL_FAIL TRUE)
  add_test(NAME reject_invalid_command COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/invalid-command.csv")
  set_tests_properties(reject_invalid_command PROPERTIES WILL_FAIL TRUE)
  add_test(NAME reject_out_of_range COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/out-of-range.csv")
  set_tests_properties(reject_out_of_range PROPERTIES WILL_FAIL TRUE)
  add_test(NAME reject_trailing_junk COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/trailing-junk.csv")
  set_tests_properties(reject_trailing_junk PROPERTIES WILL_FAIL TRUE)
  add_test(NAME reject_mismatched_result COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/mismatched-result.csv")
  set_tests_properties(reject_mismatched_result PROPERTIES WILL_FAIL TRUE)
  add_test(NAME reject_empty_suite COMMAND rpn "${CMAKE_CURRENT_SOURCE_DIR}/tests/empty-suite.csv")
  set_tests_properties(reject_empty_suite PROPERTIES WILL_FAIL TRUE)
endif()
